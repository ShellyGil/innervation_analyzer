import pandas as pd import numpy as np import matplotlib.pyplot as plt from scipy import stats from js import document, alert from pyodide.ffi import create_proxy import asyncio # --- DATA STORE --- STORE = { "CFA_L": [], "CFA_R": [], "Carr_L": [], "Carr_R": [] } # --- 1. CORE LOGIC --- async def add_files(key, input_id, badge_id): inp = document.getElementById(input_id) files = inp.files if files.length == 0: alert("Please select files first.") return count = 0 for i in range(files.length): f = files.item(i) text = await f.text() # Robust Parsing clean = text.replace(',', ' ').replace('\t', ' ') tokens = clean.split() nums = [] for t in tokens: try: nums.append(float(t)) except: pass if nums: STORE[key].append(np.mean(nums)) count += 1 # Update UI bdg = document.getElementById(badge_id) bdg.innerText = f"{len(STORE[key])}" bdg.classList.add("active") inp.value = "" def clear_group(group_prefix): # Clears both Left and Right for that group STORE[f"{group_prefix}_L"] = [] STORE[f"{group_prefix}_R"] = [] document.getElementById(f"bdg_{group_prefix}_L").innerText = "0" document.getElementById(f"bdg_{group_prefix}_L").classList.remove("active") document.getElementById(f"bdg_{group_prefix}_R").innerText = "0" document.getElementById(f"bdg_{group_prefix}_R").classList.remove("active") def run_analysis(event): # Validate Pairing if len(STORE["CFA_L"]) != len(STORE["CFA_R"]): alert(f"Mismatch in CFA!\nLeft: {len(STORE['CFA_L'])}, Right: {len(STORE['CFA_R'])}\nPlease clear and re-upload pairs.") return if len(STORE["Carr_L"]) != len(STORE["Carr_R"]): alert(f"Mismatch in Carrageenan!\nLeft: {len(STORE['Carr_L'])}, Right: {len(STORE['Carr_R'])}\nPlease clear and re-upload pairs.") return if len(STORE["CFA_L"]) == 0 or len(STORE["Carr_L"]) == 0: alert("Missing Data. Please upload files for both groups.") return # Calculate Ratios try: ratios_cfa = [ l/r for l,r in zip(STORE["CFA_L"], STORE["CFA_R"]) ] ratios_carr = [ l/r for l,r in zip(STORE["Carr_L"], STORE["Carr_R"]) ] except ZeroDivisionError: alert("Error: A Right Paw value is 0. Cannot divide by zero.") return dataset = [ratios_cfa, ratios_carr] names = ["CFA", "Carrageenan"] # User Titles title_text = document.getElementById("in-title").value ylabel_text = document.getElementById("in-ylabel").value # PLOTTING fig, ax = plt.subplots(figsize=(7, 6)) # COLORS: Light Blue (#87CEEB) and Grey (#D3D3D3) colors = ['#87CEEB', '#D3D3D3'] edge_colors = ['#4682B4', '#A9A9A9'] means = [np.mean(d) for d in dataset] sems = [stats.sem(d) if len(d)>1 else 0 for d in dataset] # Bars ax.bar(names, means, yerr=sems, capsize=8, color=colors, edgecolor=edge_colors, linewidth=1.5, width=0.6) # Scatter (Individual Mice) for i, d in enumerate(dataset): x = np.random.normal(i, 0.04, size=len(d)) ax.scatter(x, d, color='black', alpha=0.7, s=50, zorder=3) ax.spines['top'].set_visible(False) ax.spines['right'].set_visible(False) ax.set_ylabel(ylabel_text, fontsize=12, fontweight='bold') ax.set_title(title_text, fontsize=14, fontweight='bold') # Render document.getElementById("plot-area").innerHTML = "" display(fig, target="plot-area", append=False) # STATS (Unpaired T-Test) t_stat, p = stats.ttest_ind(ratios_cfa, ratios_carr) div = document.getElementById("stats-area") div.style.display = "block" res = "SIGNIFICANT *" if p < 0.05 else "ns" div.innerText = f"Unpaired T-Test Results:\n------------------------\nCFA Ratio Mean: {np.mean(ratios_cfa):.3f}\nCarr Ratio Mean: {np.mean(ratios_carr):.3f}\n\nP-Value: {p:.5f}\nResult: {res}" # --- 2. EVENT BINDING --- async def on_add_cfa_l(e): await add_files("CFA_L", "in_CFA_L", "bdg_CFA_L") async def on_add_cfa_r(e): await add_files("CFA_R", "in_CFA_R", "bdg_CFA_R") async def on_add_carr_l(e): await add_files("Carr_L", "in_Carr_L", "bdg_Carr_L") async def on_add_carr_r(e): await add_files("Carr_R", "in_Carr_R", "bdg_Carr_R") def on_clr_cfa(e): clear_group("CFA") def on_clr_carr(e): clear_group("Carr") def main(): document.getElementById("btn_add_CFA_L").addEventListener("click", create_proxy(on_add_cfa_l)) document.getElementById("btn_add_CFA_R").addEventListener("click", create_proxy(on_add_cfa_r)) document.getElementById("btn_add_Carr_L").addEventListener("click", create_proxy(on_add_carr_l)) document.getElementById("btn_add_Carr_R").addEventListener("click", create_proxy(on_add_carr_r)) document.getElementById("btn_clr_CFA").addEventListener("click", create_proxy(on_clr_cfa)) document.getElementById("btn_clr_Carr").addEventListener("click", create_proxy(on_clr_carr)) document.getElementById("btn-run").addEventListener("click", create_proxy(run_analysis)) main()
