import pandas as pd import numpy as np import matplotlib.pyplot as plt from scipy import stats import js from js import document import asyncio # --- GLOBAL STORE --- STORE = {"Control": [], "CFA": [], "Carr": []} def log(msg): log_div = document.getElementById("sys-log") log_div.innerText = f"> {msg}" # --- SYSTEM READY INDICATOR --- def set_ready(): dot = document.getElementById("status-dot") txt = document.getElementById("status-text") dot.style.backgroundColor = "#2ecc71" txt.innerText = "System Ready" txt.style.color = "#2ecc71" log("Python loaded. Ready for data.") # Call immediately set_ready() # --- FILE HANDLER --- async def add_files(group): try: log(f"Reading {group}...") inp = document.getElementById(f"in_{group}") files = inp.files if files.length == 0: js.alert("Select files first.") return count = 0 for i in range(files.length): f = files.item(i) text = await f.text() # Robust Parse clean = text.replace(',', ' ').replace('\t', ' ') tokens = clean.split() nums = [] for t in tokens: try: nums.append(float(t)) except: pass if nums: STORE[group].append(np.mean(nums)) count += 1 # Update Badge bdg = document.getElementById(f"bdg_{group}") bdg.innerText = f"{len(STORE[group])} Loaded" bdg.classList.add("active") inp.value = "" # Clear input log(f"Added {count} files to {group}.") except Exception as e: js.alert(f"Error: {e}") def clear_group(group): STORE[group] = [] bdg = document.getElementById(f"bdg_{group}") bdg.innerText = "0 Loaded" bdg.classList.remove("active") log(f"Cleared {group}.") # --- PLOT HANDLER --- def run_analysis(*args): try: log("Generating...") c = STORE["Control"] cfa = STORE["CFA"] carr = STORE["Carr"] if not (c and cfa and carr): js.alert("Missing data. Add files for all groups.") return dataset = [c, cfa, carr] names = ["Control", "CFA", "Carrageenan"] # Plot fig, ax = plt.subplots(figsize=(6, 5)) colors = ['#E0E0E0', '#FFCDD2', '#BBDEFB'] edges = ['#424242', '#D32F2F', '#1976D2'] means = [np.mean(d) for d in dataset] sems = [stats.sem(d) if len(d)>1 else 0 for d in dataset] ax.bar(names, means, yerr=sems, capsize=5, color=colors, edgecolor=edges) for i, d in enumerate(dataset): x = np.random.normal(i, 0.04, size=len(d)) ax.scatter(x, d, color='black', alpha=0.7, zorder=3) ax.spines['top'].set_visible(False) ax.spines['right'].set_visible(False) ax.set_ylabel("Innervation Index") ax.set_title("Analysis Results") # Display Plot document.getElementById("plot-target").innerHTML = "" display(fig, target="plot-target", append=False) # Display Stats f, p = stats.f_oneway(*dataset) s_div = document.getElementById("stats-target") s_div.style.display = "block" s_div.innerText = f"ANOVA P-Value: {p:.5f}\n" + ("(Significant *)" if p < 0.05 else "(Not Significant)") log("Done.") except Exception as e: js.alert(f"Plot Error: {e}")
