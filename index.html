import pandas as pd import numpy as np import matplotlib.pyplot as plt from scipy import stats from js import document, alert from pyodide.ffi import create_proxy import asyncio # --- GLOBAL STORE --- STORE = {"Control": [], "CFA": [], "Carr": []} # --- 1. CORE FUNCTIONS --- async def add_files(group_key, input_id, badge_id): inp = document.getElementById(input_id) files = inp.files if files.length == 0: alert("Select files first!") return for i in range(files.length): f = files.item(i) text = await f.text() # Robust Parse clean = text.replace(',', ' ').replace('\t', ' ') tokens = clean.split() nums = [] for t in tokens: try: nums.append(float(t)) except: pass if nums: STORE[group_key].append(np.mean(nums)) # Update UI bdg = document.getElementById(badge_id) bdg.innerText = f"{len(STORE[group_key])} Loaded" bdg.classList.add("active") inp.value = "" def clear_group(group_key, badge_id): STORE[group_key] = [] bdg = document.getElementById(badge_id) bdg.innerText = "0 Loaded" bdg.classList.remove("active") def run_analysis(event): c = STORE["Control"] cfa = STORE["CFA"] carr = STORE["Carr"] if not (c and cfa and carr): alert("Missing Data! Please add files for all groups.") return dataset = [c, cfa, carr] names = ["Control", "CFA", "Carrageenan"] # Plot fig, ax = plt.subplots(figsize=(8, 6)) colors = ['lightgray', 'firebrick', 'orange'] means = [np.mean(d) for d in dataset] sems = [stats.sem(d) if len(d)>1 else 0 for d in dataset] ax.bar(names, means, yerr=sems, capsize=8, color=colors, edgecolor='black', alpha=0.9) for i, d in enumerate(dataset): x = np.random.normal(i, 0.04, size=len(d)) ax.scatter(x, d, color='black', alpha=0.6, s=40, zorder=3) ax.spines['top'].set_visible(False) ax.spines['right'].set_visible(False) ax.set_ylabel("Innervation Index", fontsize=12, fontweight='bold') ax.set_title("Footpad Innervation Analysis", fontsize=14, fontweight='bold') document.getElementById("plot-area").innerHTML = "" display(fig, target="plot-area", append=False) # Stats f, p = stats.f_oneway(*dataset) div = document.getElementById("stats-area") div.style.display = "block" res = "SIGNIFICANT *" if p < 0.05 else "ns" div.innerText = f"ANOVA P-Value: {p:.5f}\nResult: {res}" # --- 2.
