import pandas as pd import numpy as np import matplotlib.pyplot as plt from scipy import stats from js import document, alert from pyodide.ffi import create_proxy import asyncio # --- DATA STORE --- STORE = {"Control": [], "CFA": [], "Carr": []} # --- 1. DATA HANDLING --- async def add_files(group_key, input_id, badge_id): inp = document.getElementById(input_id) files = inp.files if files.length == 0: alert("Please select files first!") return for i in range(files.length): f = files.item(i) text = await f.text() # Robust Parse clean = text.replace(',', ' ').replace('\t', ' ') tokens = clean.split() nums = [] for t in tokens: try: nums.append(float(t)) except: pass if nums: STORE[group_key].append(np.mean(nums)) # Update UI bdg = document.getElementById(badge_id) bdg.innerText = f"{len(STORE[group_key])}" bdg.classList.add("active") inp.value = "" def clear_group(group_key, badge_id): STORE[group_key] = [] bdg = document.getElementById(badge_id) bdg.innerText = "0" bdg.classList.remove("active") # --- 2. ANALYSIS & PLOTTING --- def run_analysis(event): c = STORE["Control"] cfa = STORE["CFA"] carr = STORE["Carr"] if not (c and cfa and carr): alert("Missing Data! Please add files for all 3 groups.") return dataset = [c, cfa, carr] names = ["Control", "CFA", "Carrageenan"] # Get User Settings title_text = document.getElementById("in-title").value ylabel_text = document.getElementById("in-ylabel").value plot_style = document.getElementById("in-style").value # Setup Plot fig, ax = plt.subplots(figsize=(9, 6)) colors = ['lightgray', 'firebrick', 'orange'] edge_colors = ['black', 'darkred', 'darkorange'] if plot_style == "bar": means = [np.mean(d) for d in dataset] sems = [stats.sem(d) if len(d)>1 else 0 for d in dataset] ax.bar(names, means, yerr=sems, capsize=8, color=colors, edgecolor='black', alpha=0.9) else: # Box Plot bplot = ax.boxplot(dataset, labels=names, patch_artist=True) for patch, color in zip(bplot['boxes'], colors): patch.set_facecolor(color) patch.set_edgecolor('black') # Scatter Points (Individual Mice) for i, d in enumerate(dataset): # Adjust x-pos based on plot type (Bar uses 0,1,2 - Box uses 1,2,3) x_pos = i if plot_style == "bar" else i + 1 x = np.random.normal(x_pos, 0.04, size=len(d)) ax.scatter(x, d, color='black', alpha=0.6, s=40, zorder=3) # Styling ax.spines['top'].set_visible(False) ax.spines['right'].set_visible(False) ax.set_ylabel(ylabel_text, fontsize=12, fontweight='bold') ax.set_title(title_text, fontsize=14, fontweight='bold') # Render document.getElementById("plot-area").innerHTML = "" display(fig, target="plot-area", append=False) # Stats f, p = stats.f_oneway(*dataset) div = document.getElementById("stats-area") div.style.display = "block" res = "SIGNIFICANT *" if p < 0.05 else "ns" div.innerText = f"One-Way ANOVA P-Value: {p:.5f}\nResult: {res}" # --- 3. EVENT BINDING (Wrapper Functions) --- async def on_add_ctrl(e): await add_files("Control", "file-ctrl", "bdg-ctrl") async def on_add_cfa(e): await add_files("CFA", "file-cfa", "bdg-cfa") async def on_add_carr(e): await add_files("Carr", "file-carr", "bdg-carr") def on_clr_ctrl(e): clear_group("Control", "bdg-ctrl") def on_clr_cfa(e): clear_group("CFA", "bdg-cfa") def on_clr_carr(e): clear_group("Carr", "bdg-carr") def main(): # Bind Buttons document.getElementById("btn-add-ctrl").addEventListener("click", create_proxy(on_add_ctrl)) document.getElementById("btn-add-cfa").addEventListener("click", create_proxy(on_add_cfa)) document.getElementById("btn-add-carr").addEventListener("click", create_proxy(on_add_carr)) document.getElementById("btn-clr-ctrl").addEventListener("click", create_proxy(on_clr_ctrl)) document.getElementById("btn-clr-cfa").addEventListener("click", create_proxy(on_clr_cfa)) document.getElementById("btn-clr-carr").addEventListener("click", create_proxy(on_clr_carr)) # Enable Main Button run_btn = document.getElementById("btn-run") run_btn.addEventListener("click", create_proxy(run_analysis)) run_btn.disabled = False # Set Status document.getElementById("sys-dot").classList.add("ready") document.getElementById("sys-msg").innerText = "System Ready" main()
