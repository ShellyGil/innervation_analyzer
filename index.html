import pandas as pd import numpy as np import matplotlib.pyplot as plt from scipy import stats import js from js import document import asyncio # --- GLOBAL STORE --- STORE = {"Control": [], "CFA": [], "Carr": []} def set_ready(): """Updates the little status bubble in bottom right.""" dot = document.getElementById("status-dot") txt = document.getElementById("status-text") dot.style.backgroundColor = "#2ecc71" txt.innerText = "System Ready" # Call immediately to show python is loaded set_ready() async def add_files(group): try: inp = document.getElementById(f"in_{group}") files = inp.files if files.length == 0: js.alert("Please select files first.") return count = 0 for i in range(files.length): f = files.item(i) text = await f.text() # Robust Parsing (Handles newlines, tabs, spaces, commas) clean = text.replace(',', ' ').replace('\t', ' ') tokens = clean.split() nums = [] for t in tokens: try: nums.append(float(t)) except: pass if nums: # Take mean if multiple numbers found in one file STORE[group].append(np.mean(nums)) count += 1 # Update Badge bdg = document.getElementById(f"bdg_{group}") bdg.innerText = f"{len(STORE[group])} Loaded" bdg.classList.add("active") inp.value = "" # Clear input except Exception as e: js.alert(f"Error: {e}") def clear_group(group): STORE[group] = [] bdg = document.getElementById(f"bdg_{group}") bdg.innerText = "0 Loaded" bdg.classList.remove("active") def run_analysis(*args): try: # Check Data c = STORE["Control"] cfa = STORE["CFA"] carr = STORE["Carr"] if not (c and cfa and carr): js.alert("Missing Data! Please ensure all 3 groups have files loaded.") return dataset = [c, cfa, carr] names = ["Control", "CFA", "Carrageenan"] # Plotting fig, ax = plt.subplots(figsize=(8, 6)) # ORIGINAL COLORS (As requested) # Control=Grey, CFA=Red (Inflammation), Carr=Orange (Acute) colors = ['lightgray', 'firebrick', 'orange'] means = [np.mean(d) for d in dataset] sems = [stats.sem(d) if len(d)>1 else 0 for d in dataset] # Draw Bars ax.bar(names, means, yerr=sems, capsize=8, color=colors, edgecolor='black', alpha=0.9) # Draw Scatter Points (Individual Mice) for i, d in enumerate(dataset): x = np.random.normal(i, 0.04, size=len(d)) ax.scatter(x, d, color='black', alpha=0.6, s=40, zorder=3) ax.spines['top'].set_visible(False) ax.spines['right'].set_visible(False) ax.set_ylabel("Innervation Index", fontsize=12, fontweight='bold') ax.set_title("Footpad Innervation Analysis", fontsize=14, fontweight='bold') # Render Plot document.getElementById("plot-area").innerHTML = "" display(fig, target="plot-area", append=False) # Stats Output f, p = stats.f_oneway(*dataset) stats_div = document.getElementById("stats-area") stats_div.style.display = "block" res_text = f"--- STATISTICS REPORT ---\n" res_text += f"Total Mice Analyzed: {sum(len(x) for x in dataset)}\n" res_text += f"One-Way ANOVA P-Value: {p:.5f}\n" if p < 0.05: res_text += ">> SIGNIFICANT DIFFERENCE DETECTED (*)" else: res_text += ">> NO SIGNIFICANT DIFFERENCE (ns)" stats_div.innerText = res_text except Exception as e: js.alert(f"Analysis Error: {e}")
