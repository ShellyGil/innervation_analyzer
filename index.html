import pandas as pd import numpy as np import matplotlib.pyplot as plt from scipy import stats from js import document, alert from pyodide.ffi import create_proxy import asyncio import io, base64 STORE = { "CFA_L": [], "CFA_R": [], "Carr_L": [], "Carr_R": [] } current_fig = None # --- FILE HANDLING --- async def add_files(key, input_id, badge_id): inp = document.getElementById(input_id) files = inp.files if files.length == 0: alert("Select files first.") return for i in range(files.length): f = files.item(i) text = await f.text() clean = text.replace(',', ' ').replace('\t', ' ') tokens = clean.split() nums = [] for t in tokens: try: nums.append(float(t)) except: pass if nums: STORE[key].append(np.mean(nums)) bdg = document.getElementById(badge_id) bdg.innerText = f"{len(STORE[key])}" bdg.classList.add("active") inp.value = "" def clear_group(group_prefix): STORE[f"{group_prefix}_L"] = [] STORE[f"{group_prefix}_R"] = [] document.getElementById(f"bdg_{group_prefix}_L").innerText = "0" document.getElementById(f"bdg_{group_prefix}_L").classList.remove("active") document.getElementById(f"bdg_{group_prefix}_R").innerText = "0" document.getElementById(f"bdg_{group_prefix}_R").classList.remove("active") # --- PLOTTING LOGIC --- def run_analysis(event): global current_fig # 1. Validation if len(STORE["CFA_L"]) != len(STORE["CFA_R"]): alert("Mismatch in CFA pairs!") return if len(STORE["Carr_L"]) != len(STORE["Carr_R"]): alert("Mismatch in Carrageenan pairs!") return if len(STORE["CFA_L"]) == 0 or len(STORE["Carr_L"]) == 0: alert("Missing Data!") return # 2. Get Settings mode = document.getElementById("in-mode").value title = document.getElementById("in-title").value ylabel = document.getElementById("in-ylabel").value # 3. Mode Selection if mode == "ratio": plot_ratio_graph(title, ylabel) else: plot_raw_graph(title, ylabel) # Enable Download document.getElementById("btn-download").disabled = False def plot_ratio_graph(title, ylabel): global current_fig # Calc Ratios try: r_cfa = [l/r for l,r in zip(STORE["CFA_L"], STORE["CFA_R"])] r_carr = [l/r for l,r in zip(STORE["Carr_L"], STORE["Carr_R"])] except ZeroDivisionError: alert("Error: Right Paw value is 0.") return # Stats t_stat, p = stats.ttest_ind(r_cfa, r_carr) # Plot fig, ax = plt.subplots(figsize=(6.5, 5.5)) current_fig = fig means = [np.mean(r_cfa), np.mean(r_carr)] sems = [stats.sem(r_cfa), stats.sem(r_carr)] names = ["CFA", "Carrageenan"] colors = ['#87CEEB', '#D3D3D3'] # Light Blue, Grey ax.bar(names, means, yerr=sems, capsize=8, color=colors, edgecolor='black', width=0.6) # Scatter for i, d in enumerate([r_cfa, r_carr]): x = np.random.normal(i, 0.04, size=len(d)) ax.scatter(x, d, color='black', alpha=0.6, s=50, zorder=3) # Significance Bracket if p < 0.05: y_max = max(max(r_cfa), max(r_carr)) h = y_max * 0.05 y_top = y_max + h * 1.5 ax.plot([0, 0, 1, 1], [y_top-h, y_top, y_top, y_top-h], lw=1.5, c='black') stars = "*" if p >= 0.01 else ("**" if p >= 0.001 else "***") ax.text(0.5, y_top, stars, ha='center', va='bottom', fontsize=16, fontweight='bold') ax.set_ylim(top=y_top * 1.1) ax.spines['top'].set_visible(False) ax.spines['right'].set_visible(False) ax.set_title(title, fontweight='bold') ax.set_ylabel(ylabel, fontweight='bold') document.getElementById("plot-area").innerHTML = "" display(fig, target="plot-area", append=False) # Stats Text div = document.getElementById("stats-area") div.style.display = "block" div.innerText = f"T-Test P-Value: {p:.5f} ({'Significant *' if p<0.05 else 'ns'})" def plot_raw_graph(title, ylabel): global current_fig # Prepare Data for "Long Graph" # Order: CFA Mice (R then L), then Carr Mice (R then L) labels = [] values = [] colors = [] # CFA Loop for i in range(len(STORE["CFA_R"])): # Right (Control) - Grey labels.append(f"CFA {i+1} R") values.append(STORE["CFA_R"][i]) colors.append("#D3D3D3") # Grey # Left (Injured) - Red labels.append(f"CFA {i+1} L") values.append(STORE["CFA_L"][i]) colors.append("firebrick") # Red # Carr Loop for i in range(len(STORE["Carr_R"])): # Right (Control) - Grey labels.append(f"Carr {i+1} R") values.append(STORE["Carr_R"][i]) colors.append("#D3D3D3") # Grey # Left (Injured) - Light Blue labels.append(f"Carr {i+1} L") values.append(STORE["Carr_L"][i]) colors.append("#87CEEB") # Light Blue # Dynamic Width based on number of mice total_bars = len(values) width = max(8, total_bars * 0.5) fig, ax = plt.subplots(figsize=(width, 6)) current_fig = fig x_pos = np.arange(len(labels)) ax.bar(x_pos, values, color=colors, edgecolor='black', width=0.8) ax.set_xticks(x_pos) ax.set_xticklabels(labels, rotation=45, ha='right', fontsize=9) ax.spines['top'].set_visible(False) ax.spines['right'].set_visible(False) ax.set_title(title, fontweight='bold') ax.set_ylabel(ylabel, fontweight='bold') document.getElementById("plot-area").innerHTML = "" display(fig, target="plot-area", append=False) document.getElementById("stats-area").style.display = "none" # Hide stats for raw view # --- DOWNLOAD --- def download_plot(event): if current_fig is None: return buf = io.BytesIO() current_fig.savefig(buf, format='png', dpi=300, bbox_inches='tight') buf.seek(0) img_str = "data:image/png;base64," + base64.b64encode(buf.read()).decode('UTF-8') link = document.createElement('a') link.href = img_str link.download = "innervation_graph.png" document.body.appendChild(link) link.click() document.body.removeChild(link) # --- BINDING --- async def on_add_cfa_l(e): await add_files("CFA_L", "in_CFA_L", "bdg_CFA_L") async def on_add_cfa_r(e): await add_files("CFA_R", "in_CFA_R", "bdg_CFA_R") async def on_add_carr_l(e): await add_files("Carr_L", "in_Carr_L", "bdg_Carr_L") async def on_add_carr_r(e): await add_files("Carr_R", "in_Carr_R", "bdg_Carr_R") def on_clr_cfa(e): clear_group("CFA") def on_clr_carr(e): clear_group("Carr") def main(): document.getElementById("btn_add_CFA_L").addEventListener("click", create_proxy(on_add_cfa_l)) document.getElementById("btn_add_CFA_R").addEventListener("click", create_proxy(on_add_cfa_r)) document.getElementById("btn_add_Carr_L").addEventListener("click", create_proxy(on_add_carr_l)) document.getElementById("btn_add_Carr_R").addEventListener("click", create_proxy(on_add_carr_r)) document.getElementById("btn_clr_CFA").addEventListener("click", create_proxy(on_clr_cfa)) document.getElementById("btn_clr_Carr").addEventListener("click", create_proxy(on_clr_carr)) document.getElementById("btn-run").addEventListener("click", create_proxy(run_analysis)) document.getElementById("btn-download").addEventListener("click", create_proxy(download_plot)) main()
